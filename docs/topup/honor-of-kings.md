<script>
(function () {
  var grid = document.getElementById("bt-sku-grid");
  var playerId = document.getElementById("bt-player-id");
  var totalEl = document.getElementById("bt-total");
  var offEl = document.getElementById("bt-off");
  var summaryLabel = document.getElementById("bt-summary-label");
  var buyBtn = document.getElementById("bt-buy");
  var moreBtn = document.getElementById("bt-more");
  if (!grid || !playerId || !totalEl || !buyBtn) return;

  function activeSku() { return grid.querySelector(".bt-sku.is-active"); }

  function refresh() {
    var sku = activeSku();
    if (!sku) return;
    var price = sku.getAttribute("data-price");
    var off = sku.getAttribute("data-off");
    var label = sku.getAttribute("data-label");
    totalEl.textContent = window.AceBaseCurrency
      ? AceBaseCurrency.formatFromBase(parseFloat(price), "USD")
      : "US$ " + price;
    offEl.textContent = "Save: " + off + "% OFF";
    summaryLabel.textContent = label;
    buyBtn.disabled = !playerId.value.trim();
  }

  grid.addEventListener("click", function (e) {
    var sku = e.target.closest(".bt-sku");
    if (!sku) return;
    grid.querySelectorAll(".bt-sku").forEach(function (el) { el.classList.remove("is-active"); });
    sku.classList.add("is-active");
    refresh();
  });
  playerId.addEventListener("input", refresh);
  buyBtn.addEventListener("click", function () {
    if (buyBtn.disabled) return;
    var sku = activeSku();
    var label = sku ? sku.getAttribute("data-label") : "";
    var msg = "Hi, I want proxy top-up for Honor of Kings Tokens (Global) " + label + " · UID: " + playerId.value.trim();
    if (typeof window.abOpenCrisp === "function") {
      window.abOpenCrisp(msg);
    } else if (window.$crisp) {
      window.$crisp.push(["set", "message:text", msg]);
      window.$crisp.push(["do", "chat:open"]);
    } else {
      window.location.href = "/";
    }
  });
  if (moreBtn) {
    moreBtn.addEventListener("click", function () {
      grid.classList.add("is-expanded");
      moreBtn.style.display = "none";
    });
  }
  refresh();
  document.addEventListener("acebase:currency-change", refresh);
})();
</script>
