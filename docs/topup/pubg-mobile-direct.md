<script>
(function () {
  var grid = document.getElementById("og-denom-grid");
  var userId = document.getElementById("og-user-id");
  var saveId = document.getElementById("og-save-id");
  var qtyInput = document.getElementById("og-qty");
  var totalPrice = document.getElementById("og-total-price");
  var minus = document.getElementById("og-qty-minus");
  var plus = document.getElementById("og-qty-plus");
  var buyBtn = document.getElementById("og-buy-btn");
  var currencySpan = document.querySelector(".og-pubg__price-currency");
  if (!grid || !userId || !qtyInput || !totalPrice || !buyBtn) return;

  var unit = 0;
  var storageKey = "acebase_pubg_user_id";

  try {
    var saved = localStorage.getItem(storageKey);
    if (saved) {
      userId.value = saved;
      saveId.checked = true;
    }
  } catch (e) {}

  function selectedDenom() {
    return grid.querySelector(".og-pubg__denom.is-active");
  }

  function formatTotalAmount(total) {
    if (window.AceBaseCurrency) {
      if (currencySpan) currencySpan.style.display = "none";
      return AceBaseCurrency.formatFromBase(total, "USD");
    }
    if (currencySpan) currencySpan.style.display = "";
    return total.toFixed(2);
  }

  function refreshCheckout() {
    var active = selectedDenom();
    unit = active ? parseFloat(active.getAttribute("data-price")) : 0;
    var q = Math.max(1, parseInt(qtyInput.value, 10) || 1);
    qtyInput.value = q;
    var hasId = userId.value.trim().length > 0;
    var total = unit * q;
    totalPrice.textContent =
      hasId && active ? formatTotalAmount(total) : formatTotalAmount(0);
    buyBtn.disabled = !(hasId && active && total > 0);
    buyBtn.classList.toggle("is-ready", hasId && active && total > 0);
  }

  grid.addEventListener("click", function (e) {
    var btn = e.target.closest(".og-pubg__denom");
    if (!btn) return;
    grid.querySelectorAll(".og-pubg__denom").forEach(function (el) {
      el.classList.remove("is-active");
    });
    btn.classList.add("is-active");
    refreshCheckout();
  });

  userId.addEventListener("input", function () {
    try {
      if (saveId.checked) localStorage.setItem(storageKey, userId.value.trim());
    } catch (e) {}
    refreshCheckout();
  });

  saveId.addEventListener("change", function () {
    try {
      if (saveId.checked) localStorage.setItem(storageKey, userId.value.trim());
      else localStorage.removeItem(storageKey);
    } catch (e) {}
  });

  minus.addEventListener("click", function () {
    qtyInput.value = Math.max(1, (parseInt(qtyInput.value, 10) || 1) - 1);
    refreshCheckout();
  });
  plus.addEventListener("click", function () {
    qtyInput.value = Math.min(99, (parseInt(qtyInput.value, 10) || 1) + 1);
    refreshCheckout();
  });
  qtyInput.addEventListener("input", refreshCheckout);

  buyBtn.addEventListener("click", function () {
    if (buyBtn.disabled) return;
    var active = selectedDenom();
    var label = active ? active.getAttribute("data-label") : "";
    var q = qtyInput.value;
    var uid = userId.value.trim();
    var msg = "Hi, I want PUBG Mobile UC (Global) " + label + " x" + q + " · User ID: " + uid;
    if (window.$crisp) {
      window.$crisp.push(["do", "chat:open"]);
      window.$crisp.push(["set", "message:text", msg]);
    } else {
      window.location.href = "../shop/how-to-order/";
    }
  });

  refreshCheckout();
  document.addEventListener("acebase:currency-change", refreshCheckout);
})();
</script>
